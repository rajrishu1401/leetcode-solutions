1class Solution {
2public:
3    bool isMatch(string s, string p) {
4        return hlp(s,p,0,0,false,p[0]);
5    }
6
7    bool hlp(string& s, string& p,int i,int j,bool rep,char wc){
8        //cout<<s.substr(i,s.size()-i)<<"  "<<p.substr(j,p.size()-j)<<endl;
9        if(i>=s.size()&&j>=p.size()){
10            return true;
11        }
12        
13        if(j>=p.size()){
14            if(rep){
15                if(wc=='.'){
16                    return true;
17                }
18                if(wc==s[i]){
19                    return hlp(s,p,i+1,j,rep,wc);
20                }
21                return false;
22            }
23            return false;
24        }
25        
26        if(p[j]=='*'){
27            if(hlp(s,p,i,j+1,true,wc)){
28                return true;
29            }
30            return hlp(s,p,i-1,j+1,false,p[j]);
31        }
32        if(i>=s.size()){
33            if(j!=p.size()-1){
34                if(p[j+1]=='*'){
35                    return hlp(s,p,i,j+2,false,p[j]);
36                }
37            }
38            return false;
39        }
40        if(rep){
41            if(wc==s[i]||wc=='.'){
42                if(hlp(s,p,i+1,j,rep,wc)){
43                    return true;
44                }
45            }
46            rep=false;
47        }
48        if(s[i]==p[j]){
49            if(hlp(s,p,i+1,j+1,rep,p[j])){
50                return true;
51            }
52        }
53
54        if(s[i]!=p[j]){
55            if(p[j]=='.'){
56                if(hlp(s,p,i+1,j+1,rep,p[j])){
57                    return true;
58                }
59            }else{
60                if(j==p.size()-1){
61                    return false;
62                }else if(p[j+1]=='*'){
63                    return hlp(s,p,i,j+2,false,'*');
64                }
65            }
66
67        }
68        return false;
69
70    }
71
72};
