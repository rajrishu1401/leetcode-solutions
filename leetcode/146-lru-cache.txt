25            if(m[key]!=fst){
26                m[key]->pre->next=m[key]->next;

28                    m[key]->next->pre=m[key]->pre;
29                }else{
30                    lst=m[key]->pre;
31                }
32                m[key]->next=fst;
33                fst->pre=m[key];
34                m[key]->pre=nullptr;
35                fst=m[key];

38        }
39        return -1;

63                }else{

77            }
78        }
