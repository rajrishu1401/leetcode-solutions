1/**
2 * Definition for a binary tree node.
3 * struct TreeNode {
4 *     int val;
5 *     TreeNode *left;
6 *     TreeNode *right;
7 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
8 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
9 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
10 * };
11 */
12class Solution {
13public:
14    queue<int> a;
15    bool leafSimilar(TreeNode* root1, TreeNode* root2) {
16        hlp(root1);
17        if(!hlp2(root2)||!a.empty()){
18            return false;
19        }
20        return true;
21    }
22    void hlp(TreeNode* curr){
23        bool b=true;
24        if(curr->left){
25            b=false;
26            hlp(curr->left);
27        }
28        if(curr->right){
29            b=false;
30            hlp(curr->right);
31        }
32        if(b){
33            a.push(curr->val);
34        }
35    }
36    bool hlp2(TreeNode* curr){
37        bool b=true;
38        if(curr->left){
39            b=false;
40            if(!hlp2(curr->left)){
41                return false;
42            }
43        }
44        if(curr->right){
45            b=false;
46            if(!hlp2(curr->right)){
47                return false;
48            }
49        }
50        if(b){
51            if(a.empty()||curr->val!=a.front()){
52                return false;
53            }
54            a.pop();
55        }
56        return true;
57    }
58};
