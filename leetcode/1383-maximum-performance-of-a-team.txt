1class Solution {

18            ans=max(ans,(long long)(su*q.top()[0]));

20        }
21        while(i<v.size()){
22            if(q.top()[0]<v[i][1]){
23                su-=q.top()[1];

26                ans=max(ans,((long long)v[i][1]*v[i][0]));

28                long long h=su*q.top()[0];
29                ans=max(h,ans);
30            }
31            i++;
32        }
33        while(!q.empty()){
34            long long h=su*q.top()[0];

37            q.pop();
38        }
39        return ans%mod;
40    }
