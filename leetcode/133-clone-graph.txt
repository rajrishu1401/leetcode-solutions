22class Solution {
23public:
24    Node* cloneGraph(Node* node) {
25        if(node==nullptr){
26            return nullptr;
27        }
28        unordered_map<int,Node*> m;
29        queue<Node*> q;
30        q.push(node);
31        unordered_set<int> s;
32        s.insert(1);
33        while(!q.empty()){
34            Node* cur=q.front();
35            q.pop();
36            if(!m.count(cur->val)){
37                m[cur->val]=new Node(cur->val);
38            }
39            for(Node* l:cur->neighbors){
40                if(!m.count(l->val)){
41                    m[l->val]=new Node(l->val);
42                }
43                m[cur->val]->neighbors.push_back(m[l->val]);
44                if(!s.count(l->val)){
45                    s.insert(l->val);
46                    q.push(l);
47                }
48            }
49        }
50        return m[1];
51    }
52};
