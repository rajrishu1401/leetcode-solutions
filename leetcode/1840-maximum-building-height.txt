1class Solution {
2public:
3    int maxBuilding(int n, vector<vector<int>>& restrictions) {
4        //cout<<" start"<<endl;
5        if(restrictions.size()==0){
6            return n-1;
7        }
8        int ans=0;
9        sort(restrictions.begin(),restrictions.end());
10        restrictions[0][1]=min(restrictions[0][1],restrictions[0][0]-1);
11        for(int i=1;i<restrictions.size();i++){
12            int d=restrictions[i][0]-restrictions[i-1][0];
13            d+=restrictions[i-1][1];
14            restrictions[i][1]=min(restrictions[i][1],d);
15        }
16        for(int i=restrictions.size()-2;i>=0;i--){
17            int d=restrictions[i+1][0]-restrictions[i][0];
18            d+=restrictions[i+1][1];
19            restrictions[i][1]=min(restrictions[i][1],d);
20        }
21        // for(vector<int> i:restrictions){
22        //     cout<<i[0]<<"  "<<i[1]<<endl;
23        // }
24        int d=restrictions[0][0]-1;
25        if(d>restrictions[0][1]){
26            d-=restrictions[0][1];
27            d/=2;
28            ans=d+restrictions[0][1];
29        }else{
30            ans=restrictions[0][1];
31        }
32        //cout<<"0: "<<ans<<endl;
33        int a=0;
34        for(int i=1;i<restrictions.size();i++){
35            //cout<<restrictions[i][0]<<endl;
36            d=restrictions[i][0]-restrictions[i-1][0];
37            if(restrictions[i][1]>restrictions[i-1][1]){
38                a=restrictions[i][1];
39                d+=restrictions[i-1][1];
40            }else{
41                a=restrictions[i-1][1];
42                d+=restrictions[i][1];
43            }
44            //cout<<d<<" "<<a<<endl;
45            if(d>a){
46                d-=a;
47                d/=2;
48                ans=max(ans,d+a);
49                //cout<<restrictions[i][0]<<":-> "<<d+a<<endl;
50            }else{
51                ans=max(ans,a);
52                //cout<<restrictions[i][0]<<": "<<d<<endl;
53            }
54        }
55        if(restrictions[restrictions.size()-1][0]<n){
56            d=n-restrictions[restrictions.size()-1][0];
57            ans=max(ans,d+restrictions[restrictions.size()-1][1]);
58            //cout<<d+restrictions[restrictions.size()-1][1]<<endl;
59        }
60        return ans;
61    }
62};
