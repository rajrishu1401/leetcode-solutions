1class Solution {

5            return *max_element(nums.begin(),nums.end());
6        }

9            unordered_map<int,int> mp;

11                mp[i]++;
12            }
13            for(int i=0;i<nums.size();i++){
14                if(mp[nums[i]]>1) continue;

16                    m=nums[i];
17                }
18            }
19            return m;
20        }
21        if(nums.size()==1){
22            return nums[0];
23        }
24        int a=nums[0];

26        if(a==b){
27            return -1;
28        }
29        for(int i=1;i<(nums.size()-1);i++){
30            if(a==nums[i]){
31                a=-1;
32            }
33            if(b==nums[i]){
34                b=-1;
35            }
36        }
37
38        return max(a,b);
39    }
40};
