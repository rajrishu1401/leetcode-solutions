1class Solution {
2public:
3    int minCost(vector<vector<int>>& grid) {
4        vector<vector<int>> v(grid.size(),vector<int>(grid[0].size(),INT_MAX));
5        priority_queue<vector<int>,vector<vector<int>>,greater<vector<int>>> q;
6        vector<pair<int,int>> d={{0,1},{0,-1},{1,0},{-1,0}};
7        q.push({0,0,0});
8        v[0][0]=0;
9        int ans=0;
10        while(!q.empty()){
11            vector<int> u=q.top();
12            //cout<<u[1]<<" , "<<u[2]<<" -> "<<u[0]<<endl;
13            if(u[1]==grid.size()-1&&u[2]==grid[0].size()-1){
14                return u[0];
15            }
16            q.pop();
17            for(int t=0;t<d.size();t++){
18                int i=u[1]+d[t].first;
19                int j=u[2]+d[t].second;
20                if(i>=0&&j>=0&&i<grid.size()&&j<grid[0].size()){
21                    if(grid[u[1]][u[2]]==(t+1)&&v[i][j]>u[0]){
22                        //cout<<i<<" "<<j<<endl;
23                        v[i][j]=u[0];
24                        q.push({u[0],i,j});
25                    }else if(v[i][j]>(u[0]+1)){
26                        v[i][j]=u[0]+1;
27                        q.push({u[0]+1,i,j});
28                    }
29                }
30            }
31        }
32        return -1;
33    }
34};
