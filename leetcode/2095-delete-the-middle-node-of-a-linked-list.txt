1/**
2 * Definition for singly-linked list.
3 * struct ListNode {
4 *     int val;
5 *     ListNode *next;
6 *     ListNode() : val(0), next(nullptr) {}
7 *     ListNode(int x) : val(x), next(nullptr) {}
8 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
9 * };
10 */
11class Solution {
12public:
13    ListNode* deleteMiddle(ListNode* head) {
14        if(head->next==nullptr){
15            return nullptr;
16        }
17        ListNode* slow=head;
18        ListNode* fast=head;
19        ListNode* pre=nullptr;
20        while(fast->next!=nullptr&&fast->next->next!=nullptr){
21            pre=slow;
22            slow=slow->next;
23            fast=fast->next->next;
24        }
25        if(fast->next==nullptr){
26            pre->next=slow->next;
27        }else{
28            slow->next=slow->next->next;
29        }
30        return head;
31    }
32};
