1class Solution {
2public:
3    int maxNumberOfBalloons(string text) {
4        unordered_map<char,double> m;
5        m['b']=0;
6        m['a']=0;
7        m['n']=0;
8        m['l']=0;
9        m['o']=0;
10        for(char i: text){
11            if(m.count(i)){
12                if(i=='l'||i=='o'){
13                    m[i]+=0.5;
14                }else{
15                    m[i]++;
16                }
17            }
18        }
19        int ans=INT_MAX;
20        for(auto i:m){
21            ans=min(ans,(int)i.second);
22        }
23        return ans;
24    }
25};
