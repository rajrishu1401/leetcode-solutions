8            return piles[n-1];

12        }

21            if(got>h){
22                startans=midans+1;
23            }else{
24                minans=min(minans,midans);
25                endans=midans;
26            }
27        }

29    }

39        int i=count<piles.size()?ceil((double)piles[count]/(double)ans):0;
