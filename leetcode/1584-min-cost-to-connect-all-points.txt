1class DJS{
2public:
3    vector<int> rank;
4    vector<int> parent;
5    DJS(int n){
6        rank.resize(n+1);
7        for(int i=0;i<=n;i++){
8            parent.push_back(i);
9        }
10    }
11    int par(int i){
12        if(parent[i]==i){
13            return i;
14        }
15        return parent[i]=par(parent[i]);
16    }
17    bool join(int u,int v){
18        int ua=par(u);
19        int va=par(v);
20        if(ua==va){
21            return false;
22        }
23        if(rank[ua]>rank[va]){
24            parent[va]=ua;
25        }else if(rank[ua]<rank[va]){
26            parent[ua]=va;
27        }else{
28            parent[ua]=va;
29            rank[va]++;
30        }
31        return true;
32    }
33};

35class Solution {
36public:
37    int minCostConnectPoints(vector<vector<int>>& points) {
38        vector<vector<int>> d;
39        for(int i=0;i<points.size();i++){
40            for(int j=i+1;j<points.size();j++){
41                d.push_back({abs(points[i][0]-points[j][0])+abs(points[i][1]-points[j][1]),i,j});
42            }
43        }
44        sort(d.begin(),d.end());
45        DJS h(points.size());
46        int ans=0;
47        int j=0;
48        for(int i=0;i<d.size()&&j<points.size();i++){
49            if(h.join(d[i][1],d[i][2])){
50                j++;
51                ans+=d[i][0];
52            }
53        }
54        return ans;
55    }
56};
