1/**
2 * Definition for singly-linked list.
3 * struct ListNode {
4 *     int val;
5 *     ListNode *next;
6 *     ListNode() : val(0), next(nullptr) {}
7 *     ListNode(int x) : val(x), next(nullptr) {}
8 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
9 * };
10 */
11class Solution {
12public:
13    int pairSum(ListNode* head) {
14        stack<int> s;
15        int ans=0;
16        ListNode* fst=head;
17        ListNode* sl=head;
18        while(fst!=nullptr&&fst->next!=nullptr){
19            s.push(sl->val);
20            sl=sl->next;
21            fst=fst->next->next;
22        }
23        while(sl!=nullptr){
24            ans=max(ans,sl->val+s.top());
25            sl=sl->next;
26            s.pop();
27        }
28        return ans;
29    }
30};
