1/**
2 * Definition for a binary tree node.
3 * struct TreeNode {
4 *     int val;
5 *     TreeNode *left;
6 *     TreeNode *right;
7 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
8 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
9 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
10 * };
11 */
12class Solution {
13public:
14    int s;
15    int ans=0;
16    int amountOfTime(TreeNode* root, int start) {
17        s=start;
18        hlp(root);
19        return ans;
20    }
21    pair<int,bool> hlp(TreeNode* curr){
22        if(curr==nullptr){
23            return {0,false};
24        }
25        pair<int,bool> p=hlp(curr->left);
26        pair<int,bool> c=hlp(curr->right);
27        if(curr->val==s){
28            ans=max(c.first,p.first);
29            p={0,true};
30        }else if(p.second){
31            ans=max(ans,(p.first+c.first+1));
32            p={p.first+1,true};
33        }else if(c.second){
34            ans=max(ans,(p.first+c.first+1));
35            p={c.first+1,true};
36        }else{
37            p={max(c.first,p.first)+1,false};
38        }
39        //cout<<curr->val<<" : "<<p.first<<" , "<<p.second<<endl;
40        //cout<<ans<<endl;
41        return p;
42    }
43};
