1class Solution {
2public:
3    bool closeStrings(string word1, string word2) {
4        unordered_map<int,int> n;
5        unordered_map<char,int> c;
6        for(char i:word1){
7            c[i]++;
8        }
9        for(auto i:c){
10            n[i.second]++;
11        }
12        unordered_map<char,int> c1;
13        for(char i:word2){
14            if(!c.count(i)){
15                return false;
16            }
17            c1[i]++;
18        }
19        for(auto i:c1){
20            if(!n.count(i.second)){
21                return false;
22            }
23            n[i.second]--;
24            if(n[i.second]==0) n.erase(i.second);
25        }
26        if(n.size()!=0){
27            return false;
28        }
29        return true;
30    }
31};
