1class Solution {
2public:
3    int timeRequiredToBuy(vector<int>& tickets, int k) {
4        queue<pair<int,int>> q;
5        for(int i=0;i<tickets.size();i++){
6            q.push({tickets[i],i});
7        }
8        int ans=1;
9        while(!q.empty()){
10            q.front().first--;
11            if(q.front().first==0){
12                if(q.front().second==k){
13                    return ans;
14                }
15                q.pop();
16            }else{
17                q.push(q.front());
18                q.pop();
19            }
20            ans++;
21        }
22        return ans;
23    }
24};
