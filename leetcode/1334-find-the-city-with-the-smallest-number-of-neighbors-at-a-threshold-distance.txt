1class Solution {
2public:

6        for(vector<int> i:edges){
7            if(i[2]<=dis){

10            }
11        }
12        for(int i=0;i<n;i++){

16                }

20                    }

22                    if(u>dis){

24                    }

26                }
27            }
28        }
29        for(int i=0;i<n;i++){
30            for(int j=i+1;j<n;j++){
31                if(m[i][j]<=dis){
32                    c[i]++;
33                    c[j]++;
34                }
35            }
36        }
37        int h=c[0];
38        int ans=0;
39        for(int i=1;i<n;i++){
40            if(h>=c[i]){
41                ans=i;
42                h=c[i];
43            }
44        }
45        return ans;
46    }
