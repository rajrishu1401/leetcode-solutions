1class Solution {
2public:
3    vector<unordered_set<char>> hlp;
4    vector<unordered_set<char>> box;
5    vector<unordered_set<char>> r;
6    void solveSudoku(vector<vector<char>>& board) {
7        hlp.resize(9);
8        box.resize(9);
9        r.resize(9);
10        for(int i=0;i<9;i++){
11            for(int j=0;j<9;j++){
12                if(board[i][j]!='.'){
13                    hlp[j].insert(board[i][j]);
14                    r[i].insert(board[i][j]);
15                    int u=i/3;
16                    u*=3;
17                    u+=(j/3);
18                    box[u].insert(board[i][j]);
19                }
20            }
21        }
22        hl(board,0,0);
23    }
24    bool hl(vector<vector<char>>& board,int i,int j){
25        if(i==9&&j==0){
26            return true;
27        }
28        int u=j+1;
29        int v=i;
30        if(u==9){
31            u=0;
32            v++;
33        }
34        if(board[i][j]!='.'){
35            return hl(board,v,u);
36        }
37        int b1=i/3;
38        b1*=3;
39        b1+=(j/3);
40        for(char w='1';w<='9';w++){
41            if(!hlp[j].count(w)&&!r[i].count(w)&&!box[b1].count(w)){
42                hlp[j].insert(w);
43                box[b1].insert(w);
44                r[i].insert(w);
45                board[i][j]=w;
46                if(hl(board,v,u)){
47                    return true;
48                }
49                board[i][j]='.';
50                hlp[j].erase(w);
51                box[b1].erase(w);
52                r[i].erase(w);
53            }
54        }
55        return false;
56    }
57};
