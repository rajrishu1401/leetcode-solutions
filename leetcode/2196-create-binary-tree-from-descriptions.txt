1/**
2 * Definition for a binary tree node.
3 * struct TreeNode {
4 *     int val;
5 *     TreeNode *left;
6 *     TreeNode *right;
7 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
8 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
9 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
10 * };
11 */
12class Solution {
13public:
14    TreeNode* createBinaryTree(vector<vector<int>>& descriptions) {
15        unordered_map<int,TreeNode*> m;
16        unordered_set<int> n;
17        for(int i=0;i<descriptions.size();i++){
18            if(!m.count(descriptions[i][0])){
19                m[descriptions[i][0]]=new TreeNode(descriptions[i][0]);
20                n.insert(descriptions[i][0]);
21            }
22            if(!m.count(descriptions[i][1])){
23                m[descriptions[i][1]]=new TreeNode(descriptions[i][1]);
24            }
25            n.erase(descriptions[i][1]);
26            if(descriptions[i][2]==1){
27                m[descriptions[i][0]]->left=m[descriptions[i][1]];
28            }else{
29                m[descriptions[i][0]]->right=m[descriptions[i][1]];
30            }
31        }
32        return m[*n.begin()];
33    }
34};
