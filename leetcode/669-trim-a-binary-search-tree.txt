1/**
2 * Definition for a binary tree node.
3 * struct TreeNode {
4 *     int val;
5 *     TreeNode *left;
6 *     TreeNode *right;
7 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
8 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
9 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
10 * };
11 */
12class Solution {
13public:
14    int l;
15    int h;
16    TreeNode* trimBST(TreeNode* root, int low, int high) {
17        TreeNode* head=new TreeNode(-1);
18        head->left=root;
19        l=low;
20        h=high;
21        hlp(root,head,true);
22        return head->left;
23    }
24    void hlp(TreeNode* curr,TreeNode* par,bool le){
25        if(curr==nullptr){
26            return;
27        }
28        if(curr->val<l){
29            if(le){
30                par->left=curr->right;
31                
32            }else{
33                par->right=curr->right;
34            }
35            hlp(curr->right,par,le);
36            return;
37            
38        }else if(curr->val>h){
39            if(le){
40                par->left=curr->left;
41            }else{
42                par->right=curr->left;
43            }
44            hlp(curr->left,par,le);
45            return;
46        }
47        hlp(curr->left,curr,true);
48        hlp(curr->right,curr,false);
49    }
50
51
52};
