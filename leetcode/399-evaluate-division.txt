1class DJS{
2public:
3    vector<int> rank;
4    vector<int> parent;
5    DJS(int n){
6        rank.resize(n+1);
7        parent.resize(n+1);
8        for(int i=0;i<=n;i++){
9            parent[i]=i;
10        }
11    }
12    int par(int n){
13        if(parent[n]==n){
14            return n;
15        }
16        return parent[n]=par(parent[n]);
17    }
18
19    void join(int a,int b){
20        int ua=par(a);
21        int ub=par(b);
22        if(ua==ub){
23            return;
24        }
25        if(rank[ua]>rank[ub]){
26            parent[ub]=ua;
27        }else if(rank[ua]<rank[ub]){
28            parent[ua]=ub;
29        }else{
30            parent[ua]=ub;
31            rank[ub]++;
32        }
33    }
34};
